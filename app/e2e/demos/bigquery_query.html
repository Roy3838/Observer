-- Revenue and sessions by region and month, all of 2025 SELECT s.region, DATE_TRUNC(DATE(pv.event_ts), MONTH) AS month, COUNT(DISTINCT s.session_id) AS sessions, COUNT(DISTINCT o.order_id) AS orders, ROUND(SUM(o.total_usd), 2) AS revenue_usd FROM `acme-analytics.events.page_views` pv JOIN `acme-analytics.events.sessions` s USING (session_id) LEFT JOIN `acme-analytics.marts.orders` o ON o.session_id = s.session_id WHERE pv.event_ts BETWEEN '2025-01-01' AND '2025-12-31' GROUP BY 1, 2 ORDER BY month, revenue_usd DESC;
